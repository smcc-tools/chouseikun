  - 半透明の元の色（`rgba(var(--x), a)` で使う）: `--press-rgb` `--brand-rgb` `--shadow-rgb` `--muted-rgb` `--gold-rgb` `--bg-rgb` `--bg-glow-rgb`
  - 角丸: `--vote-radius` `--nav-radius` `--nav-radius-sm` `--bulk-radius`

#### なぜこの形か

設計書は「オリーブの `:root` に新トークンを足す」としていたが、それだと近い色（例: `#F6F9F0` と `#F7FAF0`）を1つのトークンにまとめた時点でオリーブの色が変わる。
フォールバックに元の色を持たせ、新トークンをオリーブで定義しなければ、**どの箇所も元の色のまま**で、ちいかわだけ値が入る。
さらに「`var(--新トークン, X)` を `X` に戻すと、置き換え前のファイルと1文字も違わない」ことを機械的に確かめられる。ホバーなど画面の記録では捕まえにくい状態も含めて、オリーブが変わっていないことの証明になる（設計書の「全要素の色を記録して突き合わせる」より強い確かめ方）。

```js
// テーマ用のトークンが「オリーブの見た目を変えない」形で使われていることの特性化テスト。
//
// 直書きの色は var(--新トークン, 元の色) に置き換えてある。新トークンはオリーブの :root で
// 定義しないので、オリーブでは元の色（フォールバック）がそのまま使われる。
// ここが崩れる（新トークンをオリーブで定義する／フォールバックを書き忘れる／
// 色味のある直書きの色を新しく足す）とオリーブの見た目が変わるか、ちいかわに
// オリーブの色が残るので、3つの約束を固定する。
const { test } = require('node:test');
const assert = require('node:assert/strict');
const fs = require('fs');
const path = require('path');

const HTML = fs.readFileSync(path.join(__dirname, '..', '..', 'test', 'index.html'), 'utf8');

// 直書きの色を置き換えたトークン（フォールバック必須）
const REPLACED_TOKENS = [
  'deco-on-dark', 'on-dark', 'text-label', 'gold-tint', 'gold-tint-strong', 'gold-text',
  'clay-border', 'clay-tint', 'clay-text', 'brand-tint', 'brand-tint-soft', 'brand-tint-head',
  'brand-tint-hover', 'brand-hover', 'brand-disabled', 'surface-sunken', 'panel-cream',
  'row-hover', 'table-head', 'text-faint', 'text-strong', 'nav-text', 'bg-glow', 'scrollbar',
  'deco-gold', 'vote-yes', 'vote-no', 'host-badge',
  'confetti-1', 'confetti-2', 'confetti-3', 'confetti-4', 'confetti-5', 'confetti-6', 'confetti-7',
  'press-rgb', 'brand-rgb', 'shadow-rgb', 'muted-rgb', 'gold-rgb', 'bg-rgb', 'bg-glow-rgb',
  'vote-radius', 'nav-radius', 'nav-radius-sm', 'bulk-radius',
];

// どのテーマでも馴染むので直書きのままでよい色
const NEUTRAL = new Set(['#fff', '#ffffff', '#000', '#000000', '#333', '#333333', '#fafafa', '#f5f5f5',
  '#eee', '#eeeeee', '#bbb', '#ccc', '#ddd', '#999', '#666', '#888', '#444', '#222', '#111', '#555',
  '#aaa', '#f0f0f0', '#e0e0e0', '#e5e5e5']);

function block(src, opener) {
  const i = src.indexOf(opener);
  if (i < 0) return '';
  return src.slice(i, src.indexOf('}', i) + 1);
}
const OLIVE_ROOT = block(HTML, '\n:root {');

test('オリーブの :root ブロックが見つかる（以降のテストの前提）', () => {
  assert.ok(OLIVE_ROOT.includes('--green:'), 'オリーブの :root が見つからない');
});

test('新トークンはオリーブの :root で定義しない（定義すると元の色が使われなくなる）', () => {
  const defined = REPLACED_TOKENS.filter(t => OLIVE_ROOT.includes(`--${t}:`));
  assert.deepEqual(defined, []);
});

test('置き換えたトークンの var() には必ずフォールバックがある', () => {
  const pat = new RegExp(`var\\(--(${REPLACED_TOKENS.join('|')})\\s*\\)`, 'g');
  assert.deepEqual(HTML.match(pat) || [], []);
});

test('色味のある直書きの色が残っていない', () => {
  let s = HTML;
  // トークンの定義ブロック（オリーブ・ちいかわ）は直書きの色を持つのが正しい
  s = s.replace(OLIVE_ROOT, '');
  s = s.replace(block(s, ':root[data-theme="chiikawa"] {'), '');
  // データURI（CSSで描いたロゴなど）は var() が使えない
  s = s.replace(/url\(\s*["']?data:[^)]*\)/g, '');
  // var(--x, フォールバック) はフォールバックごと取り除く
  s = s.replace(/var\(--[a-z0-9-]+\s*,[^()]*\)/g, '');
  // 意図して残している行: ブラウザ上部の色・座席の色タグ（保存データ）・未使用の colorMap・
  // 明示的に許可した行（テーマの見本の色など）
  s = s.split('\n').filter(l => !/name="theme-color"|const colorMap|\{ key: 'c\d'|theme-literal-ok/.test(l)).join('\n');
  const hex = (s.match(/#[0-9a-fA-F]{6}\b|#[0-9a-fA-F]{3}\b/g) || [])
    .map(h => h.toLowerCase()).filter(h => !NEUTRAL.has(h));
  assert.deepEqual([...new Set(hex)], []);
  const rgb = [...s.matchAll(/rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)/g)]
    .map(m => `${m[1]},${m[2]},${m[3]}`)
    .filter(t => t !== '0,0,0' && t !== '255,255,255');
  assert.deepEqual([...new Set(rgb)], []);
});
```

```python
    "#efe5cd": "--deco-on-dark", "#edf1e4": "--on-dark",

def replace_once(src, old, new):
    assert src.count(old) == 1, f"想定と違う出現回数（{src.count(old)}回）: {old[:80]}"
    return src.replace(old, new)

# ── 1) 文脈で行き先が変わる色は個別に置き換える ──
s = replace_once(s, '<span style="color:#3D5A3A;font-size:1rem;">◯</span>',

for m in re.finditer(r"url\(\s*[\"']?data:[^)]*\)", s): protected.append(m.span())  # データURI
for m in re.finditer(r"[^\n]*(name=\"theme-color\"|const colorMap|\{ key: 'c\d')[^\n]*", s):
    protected.append(m.span())                                                  # 意図して残す行

    out, pos = [], 0
    for a, b in protected:
        if a < pos: continue
        out.append(fn(s[pos:a])); out.append(s[a:b]); pos = b

def fix_hex(seg):
    def rep(m):
        if m.group(1):  # stroke="#X" / fill="#X"（属性に var() は使えないので style に移す）

        tok = HEX_MAP.get(m.group(3).lower())
        return f"var({tok}, {m.group(3)})" if tok else m.group(0)
    return re.sub(r'\b(stroke|fill)="(#[0-9A-Fa-f]{6})"|(#[0-9A-Fa-f]{6})\b', rep, seg)

# ── 5) rgba の元の色 ──
def fix_rgb(seg):
    def rep(m):
        tok = RGB_MAP.get((int(m.group(3)), int(m.group(4)), int(m.group(5))))
        return f"{m.group(1)}var({tok},{m.group(2)})" if tok else m.group(0)
    return re.sub(r"(rgba?\()(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\s*)(?=[,)])", rep, seg)

s = outside(lambda seg: fix_rgb(fix_hex(seg)))
open(P, "w", encoding="utf-8").write(s)
```

```js
const { test } = require('node:test');
```

```js
const fs = require('fs');

const HTML = fs.readFileSync(path.join(__dirname, '..', '..', 'test', 'index.html'), 'utf8');

const TOKENS = Object.fromEntries([...BLOCK.matchAll(/--([a-z0-9-]+)\s*:\s*([^;]+);/g)].map(m => [m[1], m[2].trim()]));

function resolve(name, seen = new Set()) {
  const v = TOKENS[name];
  if (v === undefined || seen.has(name)) return undefined;
  seen.add(name);
  const m = /^var\(--([a-z0-9-]+)\)$/.exec(v);

function lum(hex) {
  const h = hex.replace('#', '');
```

```js
  // 画像は themes/ の下に置く。オーナーが用意し、無ければ表示しない。
  const THEMES = {
    olive: { label: 'オリーブ', metaColor: '#3D5A3A' }, // theme-literal-ok
    chiikawa: {
      label: 'ちいかわ', metaColor: '#46749F', // theme-literal-ok
      fontHref: 'https://fonts.googleapis.com/css2?family=M+PLUS+Rounded+1c:wght@500;700;800&family=Zen+Maru+Gothic:wght@400;500;700&display=swap',
      images: { header: 'chiikawa/header.png', empty: 'chiikawa/empty.png', celebrate: 'chiikawa/celebrate.png' },
    },
  };

  // 未設定・知らない値・大文字違いはオリーブ。既存イベントの移行は要らない。
  function normalizeTheme(value) {
    return value === 'chiikawa' ? 'chiikawa' : 'olive';
  }

  // 画像はリポジトリ直下の themes/ にある。promote.sh は index.html と sw.js しか本番に
  // コピーしないので test/ には置けず、test のページからは1つ上をたどる。
  function themeAssetBase(pathname) {
    return /\/test\//.test(pathname || '') ? '../themes/' : 'themes/';
  }

  function themeAssetUrl(theme, kind, pathname) {
    const images = THEMES[normalizeTheme(theme)].images;
    const rel = images && images[kind];
    return rel ? themeAssetBase(pathname) + rel : '';
  }

  let currentTheme = 'olive';

  // イベントのテーマを当てる。イベントのデータを受け取るたびに呼ばれるので、何度呼んでも同じ結果になる。
  // イベントの外（トップ・マイイベント）は ?event= の無い別の読み込みなので、ここは呼ばれずオリーブのまま。
  function applyTheme(theme) {
    const t = normalizeTheme(theme);
    currentTheme = t;
    const root = document.documentElement;
    if (t === 'olive') root.removeAttribute('data-theme');
    else root.setAttribute('data-theme', t);
    const meta = document.querySelector('meta[name="theme-color"]');
    if (meta) meta.setAttribute('content', THEMES[t].metaColor);
    // フォントはそのテーマのときだけ読む（オリーブの人に余計な通信をさせない）
    const href = THEMES[t].fontHref;
    if (href && !document.querySelector(`link[data-theme-font="${t}"]`)) {
      const link = document.createElement('link');
      link.rel = 'stylesheet';
      link.href = href;
      link.dataset.themeFont = t;
      document.head.appendChild(link);
    }
    // 次に開いたとき、データを待たずに先に当てるための記録（<head> の先読みスクリプトが読む）
    try { if (eventId) localStorage.setItem('chouseikun_theme_' + eventId, t); } catch (_) {}
  }
```
